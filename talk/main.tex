% =============================================================== постановка ===
\section{Постановка}

\begin{frame}{Хотим считать степени свободы обучения, не храня траекторию}
    \begin{block}{Что измеряем}
        \small
        \alert{Активная размерность} $\dact$ отрезка обучения --- число
        независимых направлений, вдоль которых параметры реально движутся внутри
        окна. Обычно $\dact \ll P$, где $P$ --- миллионы параметров.
    \end{block}

    \begin{block}{В чём сложность}
        \small
        Измерить $\dact$ напрямую --- значит хранить всю траекторию, чего обычный
        запуск не делает. Зато он бесплатно пишет \alert{скалярные логи}: лосс,
        норму градиента, норму параметров.\\[0.8mm]
        \textbf{Что из активной размерности можно восстановить по одному такому
        скаляру?}
    \end{block}

    \begin{block}{Зачем: grokking}
        \small
        Отложенная генерализация --- это реальное сжатие геометрии траектории,
        или просто вся траектория стала мельче? И можно ли поймать переход по
        логам, которые запуск пишет и так?
    \end{block}
\end{frame}

% =================================================================== итоги ===
\section{Итоги}

\begin{frame}{Итоги: детектор упрощения, а не измеритель размерности}
    \small
    \begin{enumerate}\setlength\itemsep{1.5mm}
        \item \textbf{Где абсолютное значение честное.} В детерминированном и
              возвратном режиме с согласованным лагом один скалярный лог считает
              активную размерность до потолка $\approx 8$. Работает схема
              «задержки $+$ соседи», а не конкретный оценщик.
        \item \textbf{В grokking'е абсолютного значения нет.} Диагностики
              отвергают обе постановки, три разных механизма дают устойчивое, но
              неверное число. Уровень читать нельзя --- только изменение внутри
              запуска.
        \item \alert{\textbf{Но как детектор упрощения метод работает.}} И у
              обобщающих запусков, и при WD $=0$ он ловит реальное событие: там
              --- временное сжатие у перехода, здесь --- скатывание в рост нормы
              с траекторией ранга 1. Ложных тревог нет; есть неразличение
              \emph{вида} упрощения.
        \item \textbf{Вид различает прямое измерение.} Сжатие у grokking'а
              временное и \emph{обращается}, у запуска без WD --- постоянное.
              Разделяют время и возврат, а не глубина.
    \end{enumerate}
\end{frame}

\begin{frame}{Ограничения, честно}
    \small
    \begin{itemize}\setlength\itemsep{1.6mm}
        \item Нужен возвратный режим и лаг, согласованный с периодом, которого лог
              не выдаёт. Выше $\approx 8$ компонент оценка только порядковая, и
              линейный $\mathrm{PR}$ по задержкам почти не хуже.
        \item Нет отбора области скейлинга --- для шумного режима это очевидное
              улучшение протокола, для транзиента бесполезное: там все двадцать
              расстояний лежат в полосе шириной 13 \%.
        \item Коллапс --- четыре запуска против двух, разделённых weight decay,
              который заодно двигает норму. Метка ставится только \emph{после}
              перехода.
        \item В полнобатчевой постановке результат не воспроизводится.
    \end{itemize}
\end{frame}

        \item У двух запусков без WD на этом окне почти нет динамического
              диапазона, так что их $\times 1$ --- не дименсиональный негатив.
    \end{look}

\begin{frame}{Резерв: что именно видит поиск соседей}

        \item Справа два режима реального обучения. $\approx 16$ на транзиенте ---
              величина на пределе исключения, а не размерность.
    \end{look}
\end{frame}
